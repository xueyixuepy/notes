- 题目
  给你两个 非空 的链表，表示两个非负的整数。它们每位数字都是按照 逆序 的方式存储的，并且每个节点只能存储 一位 数字。  
  请你将两个数相加，并以相同形式返回一个表示和的链表。
  你可以假设除了数字 0 之外，这两个数都不会以 0 开头。
  输入：l1 = \[2,4,3], l2 = \[5,6,4]
  输出：\[7,0,8]
  解释：342 + 465 = 807.

/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        ListNode* now1= l1;
        ListNode* now2= l2;
        vector<int> con;
        int sum=0;
        int sign =0;
        int tem1= 0;
        int tem2= 0;

        while(true){
            if(now1!=nullptr){
                tem1 = now1->val;
                now1= now1->next;
            }
            else{
                tem1 = 0;
            }
            if(now2!=nullptr){
                tem2 = now2->val;
                now2= now2->next;
            }
            else{
                tem2 = 0;
            }
            sum=tem1+tem2+sign;
            if(sign) sign = 0;
            sign = sum/10;
            sum = sum%10;
            con.emplace_back(sum);
            sum = 0;
            
            if(now1==nullptr&&now2==nullptr){
                break;
            }

        }
        if(sign) con.emplace_back(sign);
        return tool(con,0);

    }
    ListNode* tool(vector<int> con,int i){
        if(i== con.size()){
            return nullptr;
        }
        ListNode* temNode = new ListNode(con[i],tool(con,i+1));
        return temNode;
        
    }
};

class Solution {
    public ListNode addTwoNumbers(ListNode l1, ListNode l2) {
        ListNode dummyHead = new ListNode(-1), pre = dummyHead;
        int t = 0;
        while (l1 != null || l2 != null || t != 0) {
            if (l1 != null) {
                t += l1.val;
                l1 = l1.next;
            }
            if (l2 != null) {
                t += l2.val;
                l2 = l2.next;
            }
            pre.next = new ListNode(t % 10);
            pre = pre.next;
            t /= 10;
        }

        return dummyHead.next;
    }
}
